KamProject.Arnold1963.Theorem1Result.global_orbits {n : ℕ} {H₀ : ComplexSpace n → ℂ} {G : Set (ComplexSpace n)}
  {ρ : ℝ≥0} {κ : ℝ} {f : AnalyticPhaseFunction n G ρ} (self : Theorem1Result n H₀ G ρ κ f) (z : RealPhaseSpace n) :
  z ∈ self.goodSet →
    ∃ γ,
      torusProjection (realPartPhase (γ 0)) = z ∧
        (∀ (t : ℝ), HasDerivAt γ (hamiltonianVectorField (fun w => H₀ w.1 + f.toFun w) (γ t)) t) ∧
          (∀ (t : ℝ), complexifyPhase (realPartPhase (γ t)) = γ t) ∧
            ∀ (t : ℝ), torusProjection (realPartPhase (γ t)) ∈ self.goodSet
